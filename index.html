Integrasi Database Google Sheets

Simpan data otomatis ke Google Spreadsheet pribadi Anda

* Jika dikosongkan, aplikasi tetap berjalan lancar dan menyimpan data sementara di sesi browser Anda.

Cara Menghubungkan ke Google Sheet (3 Menit):

  1. Buka Google Sheets baru di akun Google Anda, beri nama misalnya "Keuangan Rumah Tangga".
  2. Pada menu atas, klik Ekstensi (Extensions) → Apps Script.
  3. Hapus semua kode yang ada di editor, lalu salin dan tempel kode di bawah ini:
function doPost(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getActiveSheet();
  var payload = JSON.parse(e.postData.contents);
  
  // Jika aksi adalah hapus data dari web
  if (payload.action === "delete" && payload.id) {
    var rows = sheet.getDataRange().getValues();
    var targetId = String(payload.id);
    var deleted = false;

    for (var i = rows.length - 1; i >= 1; i--) {
      var rowId = rows[i][8] ? String(rows[i][8]) : ("sheet-row-" + i);
      if (rowId === targetId) {
        sheet.deleteRow(i + 1);
        deleted = true;
        break;
      }
    }

    // Cadangan pencocokan berdasarkan keterangan & nominal jika ID tidak ditemukan
    if (!deleted && payload.keterangan) {
      for (var j = rows.length - 1; j >= 1; j--) {
        if (String(rows[j][1]) === String(payload.keterangan) && Number(rows[j][2]) === Number(payload.pendapatan)) {
          sheet.deleteRow(j + 1);
          break;
        }
      }
    }

    return ContentService.createTextOutput(JSON.stringify({ "status": "deleted" }))
      .setMimeType(ContentService.MimeType.JSON);
  }
  
  // Buat header otomatis jika baris pertama masih kosong
  if (sheet.getLastRow() === 0) {
    var headers = [
      "Tanggal", 
      "Keterangan", 
      "Pendapatan Harian (IDR)", 
      "Tunjangan Anak & Istri", 
      "Nafkah Istri", 
      "Perbaikan Kendaraan", 
      "Dana Mendesak",
      "Waktu Input",
      "ID Transaksi"
    ];
    sheet.appendRow(headers);
    var headerRange = sheet.getRange(1, 1, 1, headers.length);
    headerRange.setBackground("#059669");
    headerRange.setFontColor("#ffffff");
    headerRange.setFontWeight("bold");
    sheet.setFrozenRows(1);
  }
  
  sheet.appendRow([
    payload.tanggal,
    payload.keterangan,
    Number(payload.pendapatan) || 0,
    Number(payload.tunjanganAnakIstri) || 0,
    Number(payload.nafkahIstri) || 0,
    Number(payload.perbaikanKendaraan) || 0,
    Number(payload.danaMendesak) || 0,
    new Date().toLocaleString("id-ID"),
    payload.id || ""
  ]);

  var lastRow = sheet.getLastRow();
  sheet.getRange(lastRow, 3, 1, 5).setNumberFormat('"Rp "#,##0');
  
  return ContentService.createTextOutput(JSON.stringify({ "status": "success" }))
    .setMimeType(ContentService.MimeType.JSON);
}

function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getActiveSheet();
  var rows = sheet.getDataRange().getValues();
  var result = [];

  var summary = {
    totalPendapatan: 0,
    tunjanganAnakIstri: 0,
    nafkahIstri: 0,
    perbaikanKendaraan: 0,
    danaMendesak: 0
  };
  
  for (var i = 1; i < rows.length; i++) {
    var rawDate = rows[i][0];
    var ket = String(rows[i][1] || "");
    if (!rawDate && !ket) continue;

    // Abaikan baris ringkasan/total manual jika user mengetik TOTAL di dalam sheet
    var upperCheck = (String(rawDate) + " " + ket).toUpperCase();
    if (upperCheck.indexOf("TOTAL") !== -1 || upperCheck.indexOf("JUMLAH") !== -1) {
      continue;
    }

    var formattedDate = rawDate instanceof Date 
      ? Utilities.formatDate(rawDate, Session.getScriptTimeZone(), "yyyy-MM-dd") 
      : String(rawDate);

    var p = Number(rows[i][2]) || 0;
    var t1 = Number(rows[i][3]) || 0;
    var t2 = Number(rows[i][4]) || 0;
    var t3 = Number(rows[i][5]) || 0;
    var t4 = Number(rows[i][6]) || 0;

    summary.totalPendapatan += p;
    summary.tunjanganAnakIstri += t1;
    summary.nafkahIstri += t2;
    summary.perbaikanKendaraan += t3;
    summary.danaMendesak += t4;

    result.push({
      id: rows[i][8] ? String(rows[i][8]) : ("sheet-row-" + i),
      tanggal: formattedDate,
      keterangan: ket,
      pendapatan: p,
      tunjanganAnakIstri: t1,
      nafkahIstri: t2,
      perbaikanKendaraan: t3,
      danaMendesak: t4,
      synced: true
    });
  }
  
  return ContentService.createTextOutput(JSON.stringify({
    records: result,
    summary: summary
  })).setMimeType(ContentService.MimeType.JSON);
}
